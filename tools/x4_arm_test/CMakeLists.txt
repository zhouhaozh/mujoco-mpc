cmake_minimum_required(VERSION 3.16)
project(x4_arm_axis_test LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build and run this tool on Ubuntu, not macOS")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(X4_SDK_ROOT "/home/ubuntu/Zhaohui/x4-sdk" CACHE PATH "Authoritative X4 SDK directory")
find_package(Threads REQUIRED)
find_package(Eigen3 REQUIRED)
set(SIM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../sckd_sim")
foreach(component arm hal core)
  find_library(X4_${component}_LIB NAMES ${component} PATHS "${X4_SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_library(x4_arm_test_backend STATIC "${SIM_ROOT}/cpp/runtime/x4_arm_test_backend.cpp")
target_include_directories(x4_arm_test_backend BEFORE PRIVATE "${X4_SDK_ROOT}/include")
target_include_directories(x4_arm_test_backend PUBLIC "${SIM_ROOT}/cpp")
target_link_libraries(x4_arm_test_backend PRIVATE Eigen3::Eigen Threads::Threads
  ${X4_arm_LIB} ${X4_hal_LIB} ${X4_core_LIB})
add_executable(x4_arm_axis_test "${SIM_ROOT}/cpp/apps/x4_arm_axis_test.cpp")
target_link_libraries(x4_arm_axis_test PRIVATE x4_arm_test_backend Threads::Threads)
target_compile_definitions(x4_arm_axis_test PRIVATE X4_ARM_SDK_ROOT="${X4_SDK_ROOT}")
set_property(TARGET x4_arm_axis_test PROPERTY BUILD_RPATH "${X4_SDK_ROOT}/lib")
target_compile_options(x4_arm_test_backend PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(x4_arm_axis_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(x4_arm_remote "${CMAKE_CURRENT_SOURCE_DIR}/x4_arm_remote.cpp")
target_compile_options(x4_arm_remote PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_arm_axis_check "${SIM_ROOT}/cpp/tests/test_arm_axis_check.cc")
target_include_directories(test_arm_axis_check PRIVATE "${SIM_ROOT}/cpp" "${SIM_ROOT}/third_party/doctest")
enable_testing()
add_test(NAME arm_axis_check COMMAND test_arm_axis_check)
